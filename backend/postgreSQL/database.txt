1. Normalized -> Metadata Scheme 

Table metadata_category {
  metadata_category_id INT [pk]
  main_category TEXT
}

Table sub_category {
  sub_category_id INT [pk]
  metadata_category_id INT [ref: > metadata_category.metadata_category_id]
  sub_category TEXT
}

Table element {
  element_id INT [pk]
  sub_category_id INT [ref: > sub_category.sub_category_id]
  element_name TEXT
  element_description TEXT
  is_mandatory BOOLEAN
  allowed_ontology TEXT
}

Table metadata_values {
  value_id INT [pk]
  model_id INT
  element_id INT [ref: > element.element_id]
  field_value TEXT
  ontology TEXT 
}

Ref: "metadata_category"."metadata_category_id" < "metadata_category"."main_category"
 


metadata_category -> sub_category (1:n) 
sub_category -> element (1:n) 
element -> metadata_values (1:n)
